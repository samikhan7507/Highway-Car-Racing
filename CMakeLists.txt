cmake_minimum_required(VERSION 3.10)
project(TurboRacerCG LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(OpenGL_GL_PREFERENCE LEGACY)
find_package(OpenGL REQUIRED)
find_package(GLUT REQUIRED)
add_executable(turbo_racer src/main.cpp)
target_link_libraries(turbo_racer PRIVATE OpenGL::GL GLUT::GLUT)
if(NOT WIN32 AND NOT APPLE)
  find_package(GLU QUIET)
  target_link_libraries(turbo_racer PRIVATE m)
endif()
add_executable(selftest tests/selftest.cpp)          # headless tests (no OpenGL)
enable_testing()
add_test(NAME selftest COMMAND selftest)
